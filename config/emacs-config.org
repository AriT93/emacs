*** Straight.el
straight.el is the only package manager. Every ~use-package~ installs
through straight unless it says ~:straight nil~ (built-ins, extensions
shipped inside another package, and local-only code).

Updating: ~M-x straight-pull-all~ (or ~straight-pull-package~), restart and
check things work, then ~M-x straight-freeze-versions~ and commit
~straight-lockfile.el~. After a bad update, or to sync another machine:
~git pull~ here, then ~M-x straight-thaw-versions~.
#+begin_src elisp

#+begin_src elisp
;; gpg-agent auto-starts via its socket; no need to launch it manually.
(defun ensure-gpg-agent-running ()
  "Ping gpg-agent to ensure it is ready."
  (call-process "gpg-connect-agent" nil nil nil "updatestartuptty" "/bye"))

(add-hook 'emacs-startup-hook 'ensure-gpg-agent-running)

;; Pick the newest gpg on this machine, by absolute path.
;;
;; This config is shared across machines (Apple Silicon Mac, Intel Mac, the
;; headless Linux writer-deck, work), so the binary cannot be hard-coded.  It
;; also cannot be left to epg: `epg-config--program-alist' probes "gpg2" before
;; "gpg", which on a Mac with GPG Suite installed lands on MacGPG2.
;;
;; MacGPG2 matters because GPG Suite tracks the GnuPG 2.2 LTS branch, and its
;; gpg 2.2.44 does not emit the BEGIN_ENCRYPTION status line until after it has
;; read its stdin -- while `epg-start-encrypt' waits for that line *before*
;; sending the plaintext.  The two deadlock, so saving any .gpg file (notably
;; ~/.authinfo.gpg) hangs forever with no error.  Decryption never uses that
;; handshake, which is why reading the file always worked.  GnuPG 2.5 emits the
;; line immediately.
;;
;; Comparing versions rather than blacklisting MacGPG2 by path means this keeps
;; working if GPG Suite ever ships a newer engine, and needs no per-OS branch.
(defun ari/gpg-version (program)
  "Return the GnuPG version PROGRAM reports, or nil if it is not usable."
  (ignore-errors
    (with-temp-buffer
      (when (eq 0 (call-process program nil '(t nil) nil "--version"))
        (goto-char (point-min))
        (when (re-search-forward "^gpg (GnuPG[^)]*) \\([0-9.]+\\)" nil t)
          (match-string 1))))))

(defun ari/newest-gpg ()
  "Absolute path of the newest gpg found, or nil.
Candidates are the usual per-platform locations plus whatever is on
`exec-path'.  Duplicates collapse by `file-truename' -- /usr/local/bin/gpg
is a symlink into /usr/local/MacGPG2, so it would otherwise be probed
twice -- but the path we hand to epg is the stable one, not the resolved
one, so a `brew upgrade gnupg' does not leave us pointing into a Cellar
directory that has been replaced."
  (let (seen versioned)
    (dolist (path (list "/opt/homebrew/bin/gpg"           ; Apple Silicon brew
                        "/usr/local/bin/gpg"              ; Intel brew
                        "/usr/bin/gpg"                    ; Linux distro
                        "/usr/local/MacGPG2/bin/gpg2"     ; GPG Suite
                        (executable-find "gpg")
                        (executable-find "gpg2")))
      (when (and path (file-executable-p path))
        (let ((true (file-truename path)))
          (unless (member true seen)
            (push true seen)
            (when-let* ((v (ari/gpg-version path)))
              (push (cons path v) versioned))))))
    (car (car (sort versioned
                    (lambda (a b) (version< (cdr b) (cdr a))))))))

(when-let* ((gpg (ari/newest-gpg)))
  (setq epg--configurations nil)          ; drop any configuration epg cached
  (customize-set-variable 'epg-gpg-program gpg))
(setq auth-sources '("~/.authinfo.gpg"))
(setq auth-source-cache-expiry 3600)
(setq auth-source-debug nil)
(setq auth-source-do-cache t)

(defun ari/headless-linux-p ()
  "Non-nil on a Linux machine with no display server.
This matches the writer-deck (Ubuntu Server + kmscon, no X11/Wayland),
where gpg-agent has no GNOME Keyring / pinentry-mac integration to
fall back on and must use Emacs loopback pinentry instead."
  (and (eq system-type 'gnu/linux)
       (not (getenv "DISPLAY"))
       (not (getenv "WAYLAND_DISPLAY"))))

;; On Mac / desktop Linux, gpg-agent uses the system pinentry
;; (pinentry-mac + login keychain, or pinentry-gnome3 + GNOME Keyring),
;; so Emacs should stay out of the way. Only the headless writer-deck
;; needs loopback pinentry so Emacs can prompt in the minibuffer.
(when (ari/headless-linux-p)
  (setq epa-pinentry-mode 'loopback)
  (use-package pinentry
    :config
    (pinentry-start)))

** Ghostel
A libghostty-backed terminal. See also the =ghostel-module-compile= advice in
ari-custom.

The native module is *not* built locally: Zig 0.16.0 bundles clang 20-era
resource headers, whose =float.h= does not implement the =__need_infinity_nan=
protocol that the macOS 26+ SDK's =math.h= relies on for =INFINITY= / =NAN=.
Because zig builds its own libc++ in strict-ANSI mode (=-std=c++NN=, so
=__STRICT_ANSI__= is set), neither header defines those macros and the libc++
sub-compilation fails. ghostel pins Zig with =!= .eq → @compileError=, so
bumping Zig locally is not an option; this resolves itself when upstream moves
to a Zig that ships clang 21 headers. Until then the module comes from a
GitHub release, so =auto-install= never shells out to =zig build=.

=ghostel-module-directory= sits outside straight's tree so a package rebuild
cannot delete the module while Emacs has it mapped.
    #+BEGIN_SRC elisp
  (use-package ghostel
    :defer t
    :init

    ;; it while Emacs has it mapped.  `download' takes the prebuilt module from
    ;; the GitHub release rather than shelling out to `zig build': ghostel pins
    ;; Zig exactly, and on Zig 0.16.0 the macOS 26+ SDK breaks the bundled
    ;; libc++ build (its math.h wants the clang 21 __need_infinity_nan
    ;; protocol, which Zig's clang 20 float.h doesn't implement).

*** Compilation integration
Routes =compile=, =recompile= and =project-compile= through a real PTY, so
progress bars and colour behave as they do in an interactive shell.

This advises =compilation-start=, so it also captures =F5= and anything else
that reaches =compile= internally.

The advice has to be armed *before* the first =compile=, which rules out
=:after ghostel= -- that would leave the hijack inert until something had
already opened a terminal. Loading the native module is not worth paying for
at init either, so it arms on an idle timer, in the same spirit as the other
deferred modules here.

Guarded on the module actually being present: this config is shared with
machines that have none, and there the advice would break =compile= outright
rather than falling back.

left to bind for -- =compile=, =recompile=, =project-compile= and =F5= all

variant. =ghostel-compile= / =ghostel-recompile= stay available via =M-x=.

          (end (min limit (cdr nb/current-line))))
      (when (< start end)
        (remove-text-properties start end
                                '(invisible t display "" composition ""))
        (goto-char limit)
        t)))

** org-noter
Structured, hierarchical notes kept in an org file and synced to precise
locations in a PDF (via pdf-tools). Notes stay in the org file, not the
PDF, so this is for research/reading notes, not for producing a printed
annotated copy (see the pdf-tools and Xournal++ notes above for that).

Notes files live next to the PDF, one org file per PDF, named after the
PDF (e.g. =Foo.pdf= gets =Foo.org= alongside it), rather than a shared
per-directory notes file. org-noter always searches for a
=<pdf-basename>.org= file automatically (that check is unconditional,
not something set below); leaving =org-noter-default-notes-file-names=
empty means that is the *only* candidate name, so it won't wander up
the directory tree and silently reuse an unrelated file like a
class-wide =Notes.org= it happens to find in a parent directory (that
collision - it matching and reusing a pre-existing higher-level
=Notes.org= via `locate-dominating-file` - is what silently swallowed
notes into the wrong file before). With no matching file found
anywhere above the PDF, org-noter prompts for a name/location the
first time, defaulting to =<pdf-basename>.org= in the PDF's own
directory; later sessions on that PDF find it automatically.
=org-noter-notes-search-path= is left empty too, so the document's own
directory is the only place considered.

Start a session with =M-x org-noter= from a =pdf-view= buffer (or from an
org heading with a =NOTER_DOCUMENT= property pointing at the PDF). The
first time you do this for a given PDF, it will prompt you for a notes
file name/location - accept the document's directory as the target.
Inside a session, =i= inserts a note at point (select a region in the
PDF first to attach the note to that precise spot), =M-p=/=M-n= move
between notes, =q= closes the session.

Deliberately NOT using org-pdftools/org-noter-pdftools here: that
integration (last updated 2023) is incompatible with this org-noter
version - it hijacks location handling for every note in a pdf-view
buffer (registers into =org-noter--doc-approx-location-hook=) and
builds its own location struct, which the current org-noter core's
pretty-printer chokes on (~wrong-type-argument listp~), aborting the
note insert before anything is written - for plain notes too, not just
its "precise highlight" feature. org-noter's own bundled PDF backend
(no extra package needed) handles plain and precise pdf-view notes
fine on its own.

If notes still don't seem to save: check that the org-noter session
actually has a notes file associated (look at the mode line / frame
title for the notes buffer, and confirm that buffer has a file name and
gets written on save, e.g. via =C-x C-s= or ~auto-save~). A session
started against a scratch/unsaved org buffer will silently lose notes on
exit.
org-noter's own new-notes-file flow always also offers the PDF's literal
name (spaces and all) as a candidate alongside the snake_case suggestion
below, then separately prompts for which directory to save to - there is
no variable to suppress either prompt, so both are intercepted here via
narrowly-targeted advice on =completing-read= (matched by the exact
prompt text org-noter uses for each) so a new notes file is always
snake_case and always lands next to the PDF, with no prompting.
#+begin_src elisp
  (defun ari/org-noter-notes-name-no-spaces (document-path)
    "Suggest a snake_case notes file name for DOCUMENT-PATH.
Uses the same `ari/snake-case-string' helper (ari-custom.org) that
org-remark's notes file naming uses, so both packages produce
consistently-named companion notes files."
    (concat (ari/snake-case-string (file-name-base document-path)) ".org"))

  (defun ari/org-noter--auto-answer-new-notes-prompts (orig-fn prompt collection &rest args)
    "Auto-answer org-noter's new-notes-file prompts instead of asking:
always pick the snake_case name, and always save next to the document
(the first, and after `org-noter-notes-search-path' is empty, only
directory org-noter itself offers)."
    (cond
     ((string-prefix-p "What name do you want the notes to have?" prompt)
      (or (seq-find (lambda (name) (string-match-p "\\`[a-z0-9]+\\(_[a-z0-9]+\\)*\\.org\\'" name))
                    collection)
          (apply orig-fn prompt collection args)))
     ((string-prefix-p "Where do you want to save it?" prompt)
      (if (consp collection) (car collection) (apply orig-fn prompt collection args)))
     (t (apply orig-fn prompt collection args))))

  (use-package org-noter
    :commands org-noter
    :config
    ;; Not via `:hook': use-package appends "-hook" to any name not already
    ;; ending in it, so that registered the function on a nonexistent
    ;; `org-noter-find-additional-notes-functions-hook' and new notes files
    ;; silently kept the PDF's spaced name.
    (add-hook 'org-noter-find-additional-notes-functions #'ari/org-noter-notes-name-no-spaces)
    (setq org-noter-notes-search-path nil
          org-noter-default-notes-file-names nil
          org-noter-auto-save-last-location t
          org-noter-always-create-frame nil
          org-noter-doc-split-fraction '(0.6 . 0.4)
          ;; Capture a HIGHLIGHT property (page + region coords) on any note
          ;; taken with an active selection, so the annotated-PDF export can
          ;; mark the passage - see below for why the actual PDF-mutating
          ;; half of this feature is turned back off.
          org-noter-highlight-selected-text t)
    ;; org-noter-pdf's own handler for `org-noter-highlight-selected-text'
    ;; calls `pdf-annot-add-highlight-markup-annotation', which writes a real
    ;; annotation into the *original* PDF - not wanted here (the original
    ;; must never be touched; see the pdf-tools section). Removing this
    ;; hook keeps the HIGHLIGHT property (still stored by
    ;; `org-noter-insert-note' itself, independent of this hook) without the
    ;; write; `ari/org-noter-export-annotated-pdf' draws the mark instead,
    ;; only in the exported copy.
    (remove-hook 'org-noter--add-highlight-hook #'org-noter-pdf--highlight-location)
    ;; Precise notes (region-anchored, real position on the page) are the
    ;; default for "i" - plain page-only notes carry no position an
    ;; annotated-PDF export could use (see `ari/org-noter-export-annotated-pdf'
    ;; in ari-custom.org), so swap the usual plain/precise assignment.
    (define-key org-noter-doc-mode-map (kbd "i") #'org-noter-insert-precise-note)
    (define-key org-noter-doc-mode-map (kbd "I") #'org-noter-insert-note)
    (advice-add 'completing-read :around #'ari/org-noter--auto-answer-new-notes-prompts))
